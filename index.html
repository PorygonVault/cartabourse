-- ============================================================ -- Patch — pseudos auto-générés (format uXXXXXXXXXXXX) et -- garantie de non-réutilisation, même après changement de pseudo -- À exécuter après auth-profiles.sql, dans le SQL Editor de Supabase -- ============================================================ -- Journal de TOUS les suffixes à 12 chiffres déjà distribués, pour toujours. -- Contrairement à une simple contrainte d'unicité sur profiles.display_name -- (qui ne protège que les pseudos ACTUELLEMENT utilisés), cette table garde -- la trace même après qu'un utilisateur ait changé de pseudo personnalisé — -- le numéro reste marqué "déjà utilisé" pour toujours. CREATE TABLE IF NOT EXISTS used_auto_usernames ( suffix TEXT PRIMARY KEY, used_at TIMESTAMP NOT NULL DEFAULT now() ); -- Un pseudo doit être unique parmi tous les utilisateurs (auto-généré ou choisi) DO $$ BEGIN IF NOT EXISTS ( SELECT 1 FROM pg_constraint WHERE conname = 'profiles_display_name_unique' ) THEN ALTER TABLE profiles ADD CONSTRAINT profiles_display_name_unique UNIQUE (display_name); END IF; END $$; -- Génère un suffixe à 12 chiffres qui n'a JAMAIS été distribué CREATE OR REPLACE FUNCTION generate_unique_auto_username() RETURNS TEXT AS $$ DECLARE candidate TEXT; BEGIN LOOP candidate := lpad(floor(random() * 1000000000000)::TEXT, 12, '0'); BEGIN INSERT INTO used_auto_usernames (suffix) VALUES (candidate); RETURN 'u' || candidate; EXCEPTION WHEN unique_violation THEN -- ce numéro a déjà été utilisé un jour : on retire un autre nombre CONTINUE; END; END LOOP; END; $$ LANGUAGE plpgsql; -- Remplace la création de profil : le pseudo de départ est désormais -- au format uXXXXXXXXXXXX, modifiable ensuite par l'utilisateur lui-même -- depuis sa page de compte. CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS TRIGGER AS $$ BEGIN INSERT INTO public.profiles (id, email, display_name) VALUES (NEW.id, NEW.email, generate_unique_auto_username()); RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER; -- Le trigger existant (on_auth_user_created) pointe déjà vers cette fonction, -- pas besoin de le recréer.
